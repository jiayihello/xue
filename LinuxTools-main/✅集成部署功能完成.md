## 🔍 代码实现

### deploy_lxd_server 函数

```bash
deploy_lxd_server() {
    # 1. 检查 LXD 是否已安装
    if ! is_lxd_installed; then
        msg_error "LXD 未安装，请先选择选项 1 安装 LXD。"
        return 1
    fi
    
    # 2. 确定路径
    local script_dir="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
    local server_dir="$script_dir/../server"
    local deploy_script="$server_dir/deploy_all.sh"
    
    # 3. 检查目录和文件
    if [[ ! -d "$server_dir" ]]; then
        msg_error "未找到 server 目录"
        return 1
    fi
    
    if [[ ! -f "$deploy_script" ]]; then
        msg_error "未找到部署脚本"
        return 1
    fi
    
    # 4. 显示说明并确认
    # ... (显示步骤和时间估算)
    
    read -p "是否开始部署? [y/N]: " confirm
    if [[ ! "${confirm}" =~ ^[yY]$ ]]; then
        return
    fi
    
    # 5. 执行部署
    chmod +x "$deploy_script"
    cd "$server_dir"
    bash deploy_all.sh
    
    # 6. 显示完成信息
    # ... (显示后续操作和常用命令)
    
    # 7. 返回原目录
    cd "$script_dir"
}
```
